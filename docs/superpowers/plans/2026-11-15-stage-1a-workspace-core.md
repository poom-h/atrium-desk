# Stage 1a: Workspace Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `atrium-workspace` Rust crate that prepares a workspace, creates, finds and repairs agents, and connects them to the Inbox, routines, widgets and atrium-memory, writing files byte-identical to ClaudeDesk 3.8.1.

**Architecture:** A plain Rust library crate in a Cargo workspace (no Tauri). Golden fixtures generated by compiling ClaudeDesk's own Swift sources prove byte-for-byte compatibility on macOS. Later plans (1b: PTY crate, Tauri shell, terminals; 1c: panels, palette, bundling) consume this crate.

**Tech Stack:** Rust 2021 (stable), serde / serde_json (`preserve_order`), uuid, chrono, thiserror, pathdiff, junction (Windows), tempfile (tests); Swift 6 toolchain only for the fixture generator.

**Specs:** [stage-1 design](../specs/2026-11-15-stage-1-design.md), [WORKSPACE-SPEC.md](../../WORKSPACE-SPEC.md).

## Global Constraints

- Writes the same files and the same text as ClaudeDesk 3.8.1; agent-facing text says "ClaudeDesk" and markers keep the `claudedesk:` prefix.
- Format changes are additive only: new files or keys ClaudeDesk ignores (`.claudedesk/connections.json`, `.claudedesk/workspace.json`, `.claudedesk/backups/`).
- The crate has no Tauri dependency.
- Every agent-file change goes through `fsutil::Writer` (backup when enabled, then atomic write).
- Never overwrite a JSON file that doesn't parse as an object; report it with ClaudeDesk's wording.
- `agent.json`: uppercase UUID, `createdAt` = seconds since 2001-01-01 UTC, compact, slashes escaped.
- Windows vault deny rule: `Edit(//c/Users/me/…/**)` (drive letter lowercase).
- Skill links: relative symlinks on macOS/Linux for agents inside the workspace, absolute otherwise; directory junctions on Windows.
- ClaudeDesk checkout (read-only reference): `/Users/poomhirunwiwatkul/Documents/Code/BrainHub/ManageMyClaude`. Never edit, commit or switch branches there.
- Licence MIT.

pub enum Error {

// Usage: ClaudeDeskFixtures <output folder> <ClaudeDesk Resources folder>
let manager = FileManager.default
let out = URL(fileURLWithPath: CommandLine.arguments[1])
let resources = URL(fileURLWithPath: CommandLine.arguments[2])
let scratch = manager.temporaryDirectory.appendingPathComponent("cd-fixtures-" + UUID().uuidString)
let ws = scratch.appendingPathComponent("ws")
let exe = scratch.appendingPathComponent("bin/atrium-memory")
try? manager.removeItem(at: out)
try manager.createDirectory(at: out, withIntermediateDirectories: true)
try manager.createDirectory(at: ws, withIntermediateDirectories: true)
try manager.createDirectory(at: exe.deletingLastPathComponent(), withIntermediateDirectories: true)
manager.createFile(atPath: exe.path, contents: Data("#!/bin/sh\n".utf8), attributes: [.posixPermissions: 0o755])

try Workspace.prepare(root: ws, vaultTemplate: resources.appendingPathComponent("VaultTemplate"),
                      skillTemplates: resources.appendingPathComponent("SkillTemplates"))
try manager.copyItem(at: Routines.file(in: ws), to: out.appendingPathComponent("routines.md"))

let agent = try AgentEnvironment.create(name: "Atlas Research", purpose: "Keeps the vault tidy.", host: Workspace.agents(in: ws))
let id = agent.id.uuidString

func normalize(_ text: String) -> String {
    var result = text
    for (path, token) in [(exe.path, "{{EXE}}"), ("/private" + ws.path, "{{WS}}"), (ws.path, "{{WS}}")] {
        result = result.replacingOccurrences(of: path.replacingOccurrences(of: "/", with: "\\/"), with: token)
        result = result.replacingOccurrences(of: path, with: token)
    }
    return result.replacingOccurrences(of: id.uppercased(), with: "{{ID}}")
        .replacingOccurrences(of: id.lowercased(), with: "{{id}}")
}

let files = [("agent.json", "agent.json"), ("CLAUDE.md", "CLAUDE.md"), ("memory/MEMORY.md", "MEMORY.md"),
             (".mcp.json", "mcp.json"), (".runtime/settings.json", "settings.json"),
             (".claude/settings.local.json", "settings.local.json")]

func snapshot(_ stage: String) throws {
    let folder = out.appendingPathComponent(stage)
    try manager.createDirectory(at: folder, withIntermediateDirectories: true)
    for (source, name) in files {
        let text = try String(contentsOf: URL(fileURLWithPath: agent.path).appendingPathComponent(source), encoding: .utf8)
        try normalize(text).write(to: folder.appendingPathComponent(name), atomically: true, encoding: .utf8)
    }
}

try snapshot("created")
let skills = SkillLibrary.list(library: Workspace.skills(in: ws))
try AgentEnvironment.connectInbox(agent, inbox: Workspace.inbox(in: ws),
                                  inboxSkill: skills.first { $0.name == SkillLibrary.inboxSkillName },
                                  routines: Routines.folder(in: ws))

try MemoryTool.connect(agent, executable: exe, vault: Workspace.vault(in: ws), firstTime: true)
try snapshot("connected")
let vault2 = ws.appendingPathComponent("Vault2")
try manager.createDirectory(at: vault2, withIntermediateDirectories: true)
try MemoryTool.connect(agent, executable: exe, vault: vault2, firstTime: false)
try snapshot("migrated")

let meta: [String: String] = ["id": id, "createdAt": "\(agent.createdAt.timeIntervalSinceReferenceDate)"]
try JSONSerialization.data(withJSONObject: meta, options: [.prettyPrinted, .sortedKeys]).write(to: out.appendingPathComponent("meta.json"))

print("Fixtures written to \(out.path)")

//! JSON formatted like Foundation's JSONSerialization with `.prettyPrinted, .sortedKeys`.
use serde_json::Value;

pub fn pretty_sorted(value: &Value, escape_slashes: bool) -> String {
    let mut out = String::new();
    write(value, 0, escape_slashes, &mut out);
    out
}

pub fn string_literal(text: &str, escape_slashes: bool) -> String {
    let quoted = serde_json::to_string(text).expect("strings always serialise");
    if escape_slashes {
        quoted.replace('/', "\\/")
    } else {
        quoted
    }
}

fn indent(depth: usize, out: &mut String) {
    for _ in 0..depth {
        out.push_str("  ");
    }
}

fn write(value: &Value, depth: usize, escape: bool, out: &mut String) {
    match value {
        Value::Object(map) => {
            out.push_str("{\n");
            if map.is_empty() {
                out.push('\n');
            }
            let mut keys: Vec<&String> = map.keys().collect();
            keys.sort();
            for (index, key) in keys.iter().enumerate() {
                indent(depth + 1, out);
                out.push_str(&string_literal(key, escape));
                out.push_str(" : ");
                write(&map[*key], depth + 1, escape, out);
                if index + 1 < keys.len() {
                    out.push(',');
                }
                out.push('\n');
            }
            indent(depth, out);
            out.push('}');
        }
        Value::Array(items) => {
            out.push_str("[\n");
            if items.is_empty() {
                out.push('\n');
            }
            for (index, item) in items.iter().enumerate() {
                indent(depth + 1, out);
                write(item, depth + 1, escape, out);
                if index + 1 < items.len() {
                    out.push(',');
                }
                out.push('\n');
            }
            indent(depth, out);
            out.push(']');
        }
        Value::String(text) => out.push_str(&string_literal(text, escape)),
        other => out.push_str(&other.to_string()),
    }
}

/// Writes through a temporary file in the same folder, then renames it into place.
pub fn write_atomic(path: &Path, bytes: &[u8]) -> Result<()> {
    let dir = path
        .parent()
        .ok_or_else(|| Error::Message(format!("No parent folder for {}", path.display())))?;
    fs::create_dir_all(dir)?;
    let name = path.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
    let temp = dir.join(format!(".{name}.atrium-{}", uuid::Uuid::new_v4()));
    {
        let mut file = fs::File::create(&temp)?;
        file.write_all(bytes)?;
        file.sync_all()?;
    }
    if let Err(error) = fs::rename(&temp, path) {
        let _ = fs::remove_file(&temp);
        return Err(error.into());
    }
    Ok(())
}

/// Copies a folder tree. Symbolic links are copied as the files they point to.
pub fn copy_dir(from: &Path, to: &Path) -> Result<()> {
    fs::create_dir_all(to)?;
    for entry in fs::read_dir(from)? {
        let entry = entry?;
        let target = to.join(entry.file_name());
        if entry.path().is_dir() {
            copy_dir(&entry.path(), &target)?;
        } else {
            fs::copy(entry.path(), &target)?;
        }
    }
    Ok(())
}

/// Copies of agent files taken before Atrium changes them (stage-1 design, K4).
pub struct Backups {
    root: PathBuf,
}

impl Backups {
    pub fn new(ws: &Workspace, day: &str) -> Self {
        Self { root: ws.backups().join(day) }
    }

    /// Keeps the first version of the day: a later save of the same file does nothing.
    pub fn save(&self, agent_dir: &Path, file: &Path) -> Result<()> {
        if !file.is_file() {
            return Ok(());
        }
        let relative = file
            .strip_prefix(agent_dir)
            .map_err(|_| Error::Message(format!("{} is outside {}", file.display(), agent_dir.display())))?;
        let folder = agent_dir.file_name().unwrap_or_default();
        let destination = self.root.join(folder).join(relative);
        if destination.exists() {
            return Ok(());
        }
        fs::create_dir_all(destination.parent().expect("has a parent"))?;
        fs::copy(file, &destination)?;
        Ok(())
    }

    /// Deletes day folders (named YYYY-MM-DD) older than `keep_days`.
    pub fn prune(ws: &Workspace, today: chrono::NaiveDate, keep_days: i64) -> Result<()> {
        let Ok(entries) = fs::read_dir(ws.backups()) else { return Ok(()) };
        for entry in entries.flatten() {
            let name = entry.file_name().to_string_lossy().into_owned();
            if let Ok(day) = chrono::NaiveDate::parse_from_str(&name, "%Y-%m-%d") {
                if (today - day).num_days() > keep_days {
                    fs::remove_dir_all(entry.path())?;
                }
            }
        }
        Ok(())
    }
}

/// The only way the crate changes agent files: backup (when enabled), then atomic write.
pub struct Writer {
    backups: Option<Backups>,
}

impl Writer {
    pub fn plain() -> Self {
        Self { backups: None }
    }
    pub fn with_backups(backups: Backups) -> Self {
        Self { backups: Some(backups) }
    }
    pub fn write(&self, agent_dir: &Path, file: &Path, text: &str) -> Result<()> {
        if let Some(backups) = &self.backups {
            backups.save(agent_dir, file)?;
        }
        write_atomic(file, text.as_bytes())
    }
}

use std::fs;
use std::path::{Path, PathBuf};

use crate::fsutil::{copy_dir, write_atomic};
use crate::paths::{path_text, Workspace, VAULT_FOLDERS};
use crate::{fail, Result};

/// The routines template ClaudeDesk writes, byte for byte (copied from the golden fixtures).
pub const ROUTINES_TEMPLATE: &str = include_str!("../templates/routines.md");

pub struct Templates {
    pub vault: Option<PathBuf>,
    pub skills: Option<PathBuf>,
}

/// WORKSPACE-SPEC 1.1: creates whatever is missing and never changes existing files.
/// Returns true when the workspace folder had no entries before.
pub fn prepare(ws: &Workspace, templates: &Templates) -> Result<bool> {
    if !ws.root.is_dir() {
        return fail(format!("The workspace folder is unavailable: {}", path_text(&ws.root)));
    }
    let was_empty = fs::read_dir(&ws.root)?.next().is_none();
    for folder in [ws.agents(), ws.skills(), ws.inbox(), ws.inbox_archive(), ws.logs(), ws.routines(), ws.queue(), ws.runs()] {
        if !folder.exists() {
            fs::create_dir_all(&folder)?;
        }
    }
    if !ws.routines_file().exists() {
        write_atomic(&ws.routines_file(), ROUTINES_TEMPLATE.as_bytes())?;
    }
    if let Some(skills) = &templates.skills {
        for entry in fs::read_dir(skills)?.flatten() {
            let name = entry.file_name();
            if name.to_string_lossy().starts_with('.') {
                continue;
            }
            let destination = ws.skills().join(&name);
            if !destination.exists() {
                copy_dir(&entry.path(), &destination)?;
            }
        }
    }
    let vault = ws.vault();
    if vault.exists() {
        return Ok(was_empty);
    }
    let staging = ws.root.join(format!(".creating-vault-{}", uuid::Uuid::new_v4().to_string().to_uppercase()));
    let built = build_vault(&staging, &vault, templates.vault.as_deref());
    if built.is_err() {
        let _ = fs::remove_dir_all(&staging);
    }
    built.map(|_| was_empty)
}

fn build_vault(staging: &Path, vault: &Path, template: Option<&Path>) -> Result<()> {
    match template {
        Some(template) => {
            copy_dir(template, staging)?;
            fill_placeholders(staging, &path_text(vault))?;
        }
        None => fs::create_dir(staging)?,
    }
    for folder in VAULT_FOLDERS {
        fs::create_dir_all(staging.join(folder))?;
    }
    fs::rename(staging, vault)?;
    Ok(())
}

fn fill_placeholders(folder: &Path, vault_path: &str) -> Result<()> {
    for entry in fs::read_dir(folder)?.flatten() {
        let path = entry.path();
        if path.is_dir() {
            fill_placeholders(&path, vault_path)?;
        } else if path.extension().is_some_and(|e| e == "md") {
            let text = fs::read_to_string(&path)?;
            if text.contains("{{VAULT_PATH}}") {
                write_atomic(&path, text.replace("{{VAULT_PATH}}", vault_path).as_bytes())?;
            }
        }
    }
    Ok(())
}

use std::fs;
use std::io::ErrorKind;
use std::path::Path;

use serde::Serialize;
use serde_json::{Map, Value};

use crate::applejson::pretty_sorted;
use crate::{Error, Result};

/// The .mcp.json ClaudeDesk writes for a new agent.
pub const INITIAL_MCP: &str = "{\n  \"mcpServers\": {}\n}\n";

#[derive(Clone, Debug, PartialEq)]
pub enum Style {
    ClaudeDesk { escape_slashes: bool },
    Preserve { indent: String, crlf: bool, trailing_newline: bool },
}

#[derive(Clone, Debug)]
pub struct JsonDoc {
    pub map: Map<String, Value>,
    pub style: Style,
}

impl Default for JsonDoc {
    fn default() -> Self {
        Self::new()
    }
}

impl JsonDoc {
    /// A file that doesn't exist yet: written in ClaudeDesk's form.
    pub fn new() -> Self {
        Self { map: Map::new(), style: Style::ClaudeDesk { escape_slashes: false } }
    }

    pub fn render(&self) -> String {
        let value = Value::Object(self.map.clone());
        match &self.style {
            Style::ClaudeDesk { escape_slashes } => pretty_sorted(&value, *escape_slashes),
            Style::Preserve { indent, crlf, trailing_newline } => {
                let mut buffer = Vec::new();
                let formatter = serde_json::ser::PrettyFormatter::with_indent(indent.as_bytes());
                let mut serializer = serde_json::Serializer::with_formatter(&mut buffer, formatter);
                value.serialize(&mut serializer).expect("values always serialise");
                let mut text = String::from_utf8(buffer).expect("serde_json writes UTF-8");
                if *trailing_newline {
                    text.push('\n');
                }
                if *crlf {
                    text = text.replace('\n', "\r\n");
                }
                text
            }
        }
    }
}

/// Ok(None) when the file doesn't exist; Err(NotAJsonObject) when it isn't a JSON object.
pub fn read(path: &Path) -> Result<Option<JsonDoc>> {
    let text = match fs::read_to_string(path) {
        Ok(text) => text,
        Err(error) if error.kind() == ErrorKind::NotFound => return Ok(None),
        Err(error) => return Err(error.into()),
    };
    let Ok(Value::Object(map)) = serde_json::from_str::<Value>(&text) else {
        return Err(Error::NotAJsonObject);
    };
    let style = detect_style(&text, &map);
    Ok(Some(JsonDoc { map, style }))
}

fn detect_style(text: &str, map: &Map<String, Value>) -> Style {
    let value = Value::Object(map.clone());
    if text.trim() == "{}" || text == INITIAL_MCP || text == pretty_sorted(&value, false) {
        return Style::ClaudeDesk { escape_slashes: false };
    }
    if text == pretty_sorted(&value, true) {
        return Style::ClaudeDesk { escape_slashes: true };
    }
    let crlf = text.contains("\r\n");
    let unix = text.replace("\r\n", "\n");
    let indent = unix
        .lines()
        .nth(1)
        .map(|line| line.chars().take_while(|c| *c == ' ' || *c == '\t').collect::<String>())
        .filter(|indent| !indent.is_empty())
        .unwrap_or_else(|| "  ".to_string());
    Style::Preserve { indent, crlf, trailing_newline: unix.ends_with('\n') }
}

pub fn create_agent(host: &Path, name: &str, purpose: &str, presets: &Presets, id: Uuid, created_at: f64) -> Result<AgentProfile> {
    let name = name.trim();
    if name.is_empty() {
        return fail("Give the agent a name.");
    }
    if !host.is_dir() {
        return fail("The agent host folder is unavailable. Choose a folder in Settings.");
    }
    let root = host.join(folder_name_for(name, id));
    let staging = host.join(format!(".creating-{}", id.to_string().to_uppercase()));
    let profile = AgentProfile { id, name: name.to_string(), purpose: purpose.to_string(), path: root.clone(), created_at };
    fs::create_dir(&staging)?;
    let built = (|| -> Result<()> {
        for child in REQUIRED_DIRS {
            fs::create_dir_all(staging.join(child))?;
        }
        for (relative, text) in initial_files(&profile, presets) {
            write_atomic(&staging.join(relative), text.as_bytes())?;

        write_atomic(&staging.join("agent.json"), profile.to_json().as_bytes())?;

            use std::os::unix::fs::PermissionsExt;
            fs::set_permissions(&staging, fs::Permissions::from_mode(0o700))?;
        }
        fs::rename(&staging, &root)?;
        Ok(())
    })();
    if built.is_err() {
        let _ = fs::remove_dir_all(&staging);
    }
    built.map(|_| profile)
}

use std::fs;
use std::path::{Path, PathBuf};

use crate::paths::path_text;
use crate::{fail, Result};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Attachment {
    Attached,
    Detached,
    Conflict,
}

fn link_name(skill_dir: &Path) -> std::ffi::OsString {
    skill_dir.file_name().expect("a skill folder has a name").to_os_string()
}

/// Symlinks and Windows junctions both count as links (std reports both through `is_symlink`).
fn is_link(path: &Path) -> bool {
    fs::symlink_metadata(path).map(|m| m.file_type().is_symlink()).unwrap_or(false)
}

pub fn attachment(skill_dir: &Path, skills_folder: &Path) -> Attachment {
    let link = skills_folder.join(link_name(skill_dir));
    if fs::symlink_metadata(&link).is_err() {
        return Attachment::Detached;
    }
    if !is_link(&link) {
        return Attachment::Conflict;
    }
    match (fs::canonicalize(&link), fs::canonicalize(skill_dir)) {
        (Ok(a), Ok(b)) if a == b => Attachment::Attached,
        _ => Attachment::Conflict,
    }
}

/// Links `skills_folder/<name>` to the library skill. `relative` asks for a relative symlink
/// (macOS and Linux, agents inside the workspace); Windows always uses a junction.
pub fn attach(skill_dir: &Path, skills_folder: &Path, relative: bool) -> Result<()> {
    let name = link_name(skill_dir);
    match attachment(skill_dir, skills_folder) {
        Attachment::Attached => return Ok(()),
        Attachment::Conflict => {
            return fail(format!(
                "{} already has a different skill named {}. Rename or remove it first.",
                path_text(skills_folder),
                name.to_string_lossy()
            ))
        }
        Attachment::Detached => {}
    }
    fs::create_dir_all(skills_folder)?;
    let link = skills_folder.join(&name);
    #[cfg(unix)]
    {
        let target = if relative {
            pathdiff::diff_paths(skill_dir, skills_folder).unwrap_or_else(|| skill_dir.to_path_buf())
        } else {
            skill_dir.to_path_buf()
        };
        std::os::unix::fs::symlink(target, &link)?;
    }
    #[cfg(windows)]
    {
        let _ = relative;
        junction::create(skill_dir, &link)?;
    }
    Ok(())
}

/// Links in a skills folder whose target no longer exists (stage-1 design S3).
pub fn dangling(skills_folder: &Path) -> Vec<PathBuf> {
    let Ok(entries) = fs::read_dir(skills_folder) else { return Vec::new() };
    let mut found: Vec<PathBuf> = entries
        .flatten()
        .map(|e| e.path())
        .filter(|p| is_link(p) && fs::canonicalize(p).is_err())
        .collect();
    found.sort();
    found
}

use std::path::Path;

pub const ROUTINES_MARKER: &str = "<!-- claudedesk:routines -->";
pub const WIDGETS_MARKER: &str = "<!-- claudedesk:widgets -->";

pub fn inbox_section(inbox: &Path) -> String {
    format!(
        "\n## Inbox\n{INBOX_MARKER}\nSend the user non-urgent requests, questions, reports, and proposed skills through the ClaudeDesk Inbox at `{}`, following the `{INBOX_SKILL}` skill.\n",
        path_text(inbox)
    )

pub fn routines_section(routines_file: &Path) -> String {
    format!(
        "\n## Routines\n{ROUTINES_MARKER}\nScheduled routines for every agent are defined in `{}`. Read the `{INBOX_SKILL}` skill before adding or changing one; new or riskier routines wait for the user's approval.\n",
        path_text(routines_file)
    )
}

pub fn widgets_section(folder: &Path) -> String {
    format!(

        path_text(folder)
    )
}

/// Appends each (marker, section) whose marker is missing from CLAUDE.md; writes once if anything changed.
pub fn append_sections(writer: &Writer, agent: &AgentProfile, sections: &[(&str, String)]) -> Result<()> {
    let file = agent.path.join("CLAUDE.md");
    let original = match fs::read_to_string(&file) {
        Ok(text) => text,
        Err(error) if error.kind() == std::io::ErrorKind::NotFound => String::new(),
        Err(error) => return Err(error.into()),
    };
    let mut text = original.clone();
    for (marker, section) in sections {
        if !text.contains(marker) {
            if !text.is_empty() && !text.ends_with('\n') {
                text.push('\n');
            }
            text.push_str(section);
        }
    }
    if text != original {
        writer.write(&agent.path, &file, &text)?;
    }
    Ok(())
}

fn attach_library_skill(ws: &Workspace, agent: &AgentProfile, name: &str) -> Result<()> {
    let skill = ws.skills().join(name);
    if !skill.join("SKILL.md").is_file() {
        return Ok(());
    }
    let relative = agent.path.starts_with(&ws.root);
    skills::attach(&skill, &agent.path.join(".claude/skills"), relative)
}

/// WORKSPACE-SPEC 3.5. A non-object `permissions` or `additionalDirectories` is replaced, as ClaudeDesk does.
pub fn connect_inbox(writer: &Writer, ws: &Workspace, agent: &AgentProfile) -> Result<()> {
    let settings_file = agent.path.join(".claude/settings.local.json");
    let mut doc = match jsonedit::read(&settings_file) {
        Ok(Some(doc)) => doc,
        Ok(None) => JsonDoc::new(),
        Err(Error::NotAJsonObject) => {
            return fail(format!(
                "{}'s .claude/settings.local.json is not a JSON object, so the Inbox folder was not added.",
                agent.name
            ))
        }
        Err(error) => return Err(error),
    };
    let wanted = [path_text(&ws.inbox()), path_text(&ws.routines())];
    let permissions = doc.map.entry("permissions").or_insert_with(|| json!({}));
    if !permissions.is_object() {
        *permissions = json!({});
    }
    let directories = permissions.as_object_mut().expect("object").entry("additionalDirectories").or_insert_with(|| json!([]));
    if !directories.is_array() {
        *directories = json!([]);
    }
    let list = directories.as_array_mut().expect("array");
    let missing: Vec<String> = wanted.iter().filter(|p| !list.iter().any(|v| v.as_str() == Some(p.as_str()))).cloned().collect();
    if !missing.is_empty() {
        list.extend(missing.into_iter().map(Value::String));
        writer.write(&agent.path, &settings_file, &doc.render())?;
    }
    append_sections(writer, agent, &[(INBOX_MARKER, inbox_section(&ws.inbox())), (ROUTINES_MARKER, routines_section(&ws.routines_file()))])?;
    attach_library_skill(ws, agent, INBOX_SKILL)
}

/// WORKSPACE-SPEC 3.6.
pub fn connect_widgets(writer: &Writer, ws: &Workspace, agent: &AgentProfile) -> Result<()> {
    let folder = agent.path.join("widgets");
    fs::create_dir_all(folder.join("assets"))?;
    append_sections(writer, agent, &[(WIDGETS_MARKER, widgets_section(&folder))])?;
    attach_library_skill(ws, agent, WIDGET_SKILL)
}

/// Claude Code's absolute-path rule. Windows paths use the POSIX form `//c/…` (Claude Code permissions docs).
pub fn rule_for(path: &str, windows: bool) -> String {
    if !windows {
        return format!("Edit(/{path}/**)");
    }
    let trimmed = path.strip_prefix(r"\\?\").unwrap_or(path).replace('\\', "/");
    let bytes = trimmed.as_bytes();
    let posix = if bytes.len() >= 2 && bytes[1] == b':' && bytes[0].is_ascii_alphabetic() {
        format!("/{}{}", (bytes[0] as char).to_ascii_lowercase(), &trimmed[2..])
    } else {
        trimmed
    };
    format!("Edit(/{posix}/**)")
}

pub fn vault_edit_rule(vault: &Path) -> String {
    rule_for(&path_text(vault), cfg!(windows))
}

pub fn is_app_written(entry: &Value) -> bool {
    let args_ok = entry.get("args").and_then(Value::as_array).is_some_and(|a| a.len() == 1 && a[0] == "mcp");

}

/// Adds the entry when missing. Repairs an app-written entry only when its command is gone
/// (then points it at `exe`) or its vault differs (then updates ATRIUM_MEMORY_PATH).
pub fn connect_server(writer: &Writer, agent: &AgentProfile, exe: &Path, vault: &Path) -> Result<Change> {
    let file = mcp_file(agent);
    let mut doc = match jsonedit::read(&file) {
        Ok(Some(doc)) => doc,
        Ok(None) => JsonDoc::new(),
        Err(Error::NotAJsonObject) => {
            return fail(format!("{}'s .mcp.json is not a JSON object, so the Memory Vault tools were not added.", agent.name))
        }
        Err(error) => return Err(error),
    };
    let change = {
        let servers = doc.map.entry("mcpServers").or_insert_with(|| json!({}));
        let Some(servers) = servers.as_object_mut() else {
            return fail(format!(
                "{}'s .mcp.json has a mcpServers value that is not an object, so the Memory Vault tools were not added.",
                agent.name
            ));
        };
        match servers.get_mut(SERVER_NAME) {
            Some(entry) => {
                if !is_app_written(entry) {
                    return Ok(Change::Unchanged);
                }
                let command_exists = entry.get("command").and_then(Value::as_str).is_some_and(|c| Path::new(c).is_file());
                let vault_text = path_text(vault);
                let vault_ok =
                    entry.get("env").and_then(|e| e.get("ATRIUM_MEMORY_PATH")).and_then(Value::as_str) == Some(vault_text.as_str());
                if command_exists && vault_ok {
                    return Ok(Change::Unchanged);
                }
                let object = entry.as_object_mut().expect("app-written entries are objects");
                if !command_exists {
                    object.insert("command".into(), json!(path_text(exe)));
                }
                let env = object.entry("env").or_insert_with(|| json!({}));
                if !env.is_object() {
                    *env = json!({});
                }
                env.as_object_mut().expect("object").insert("ATRIUM_MEMORY_PATH".into(), json!(vault_text));
                Change::Repaired
            }
            None => {
                servers.insert(SERVER_NAME.into(), server_entry(exe, vault, &agent.name));
                Change::Added
            }
        }
    };
    writer.write(&agent.path, &file, &doc.render())?;
    Ok(change)
}

/// The vault an app-written entry points at; None when there is no such entry or the file is unusable.
pub fn app_written_vault(agent: &AgentProfile) -> Result<Option<String>> {
    let doc = match jsonedit::read(&mcp_file(agent)) {
        Ok(Some(doc)) => doc,
        Ok(None) | Err(Error::NotAJsonObject) => return Ok(None),
        Err(error) => return Err(error),
    };
    let entry = doc.map.get("mcpServers").and_then(|s| s.get(SERVER_NAME));
    Ok(entry
        .filter(|e| is_app_written(e))
        .and_then(|e| e.get("env"))
        .and_then(|env| env.get("ATRIUM_MEMORY_PATH"))
        .and_then(Value::as_str)
        .map(str::to_string))
}

pub fn is_connected(agent: &AgentProfile) -> bool {

}

fn add_to_list(map: &mut Map<String, Value>, key: &str, value: &str, problem: &str) -> Result<bool> {
    let list = map.entry(key).or_insert_with(|| json!([]));
    let Some(items) = list.as_array_mut() else { return fail(problem) };
    if items.iter().any(|v| !v.is_string()) {
        return fail(problem);
    }
    if items.iter().any(|v| v.as_str() == Some(value)) {
        return Ok(false);
    }
    items.push(json!(value));
    Ok(true)
}

/// Approves the server, allows its tools and blocks direct vault edits. Only adds.
/// ClaudeDesk always writes `permissions` back, so a file without it gains one.
pub fn connect_settings(writer: &Writer, agent: &AgentProfile, vault: &Path) -> Result<()> {
    let file = settings_file(agent);
    let mut doc = match jsonedit::read(&file) {
        Ok(Some(doc)) => doc,
        Ok(None) => JsonDoc::new(),
        Err(Error::NotAJsonObject) => {
            return fail(format!(
                "{}'s .claude/settings.local.json is not a JSON object, so the Memory Vault tools were not approved.",
                agent.name
            ))
        }
        Err(error) => return Err(error),
    };
    if doc.map.get("permissions").is_some_and(|p| !p.is_object()) {
        return fail(format!(
            "{}'s .claude/settings.local.json has a permissions value that is not an object, so the Memory Vault tools were not approved.",
            agent.name
        ));
    }
    let list_problem = format!(
        "{}'s .claude/settings.local.json has a permission list that is not a list of strings, so the Memory Vault tools were not approved.",
        agent.name
    );
    let had_permissions = doc.map.contains_key("permissions");
    let mut changed = add_to_list(&mut doc.map, "enabledMcpjsonServers", SERVER_NAME, &list_problem)?;
    let permissions = doc.map.entry("permissions").or_insert_with(|| json!({})).as_object_mut().expect("checked above");
    changed |= add_to_list(permissions, "allow", TOOL_PERMISSION, &list_problem)?;
    changed |= add_to_list(permissions, "deny", &vault_edit_rule(vault), &list_problem)?;
    if changed || !had_permissions {
        writer.write(&agent.path, &file, &doc.render())?;
    }
    Ok(())
}

pub fn migrate_settings(writer: &Writer, agent: &AgentProfile, old: &Path, new: &Path) -> Result<()> {
    let file = settings_file(agent);
    let moved = |what: &str| format!("{}'s .claude/settings.local.json {what}, so the edit rule was not moved to the new vault.", agent.name);
    let mut doc = match jsonedit::read(&file) {
        Ok(Some(doc)) => doc,
        Ok(None) => return Ok(()),
        Err(Error::NotAJsonObject) => return fail(moved("is not a JSON object")),
        Err(error) => return Err(error),
    };
    let Some(permissions) = doc.map.get_mut("permissions") else { return Ok(()) };
    let Some(permissions) = permissions.as_object_mut() else {
        return fail(moved("has a permissions value that is not an object"));
    };
    let Some(deny) = permissions.get_mut("deny") else { return Ok(()) };
    let Some(deny) = deny.as_array_mut().filter(|d| d.iter().all(Value::is_string)) else {
        return fail(moved("has a permission list that is not a list of strings"));
    };
    let (old_rule, new_rule) = (vault_edit_rule(old), vault_edit_rule(new));
    let Some(index) = deny.iter().position(|v| v.as_str() == Some(old_rule.as_str())) else { return Ok(()) };
    if deny.iter().any(|v| v.as_str() == Some(new_rule.as_str())) {
        deny.remove(index);
    } else {
        deny[index] = json!(new_rule);
    }
    writer.write(&agent.path, &file, &doc.render())
}

pub fn migrate_instructions(writer: &Writer, agent: &AgentProfile, old: &Path, new: &Path) -> Result<()> {
    let file = agent.path.join("CLAUDE.md");
    let Ok(text) = fs::read_to_string(&file) else { return Ok(()) };
    let old_section = instructions_section(old);
    if !text.contains(&old_section) {
        return Ok(());
    }
    writer.write(&agent.path, &file, &text.replacen(&old_section, &instructions_section(new), 1))
}

/// WORKSPACE-SPEC 3.7 with the stage-1 repair rule. Migration runs before the entry is rewritten.
pub fn connect(writer: &Writer, agent: &AgentProfile, exe: &Path, vault: &Path, first_time: bool) -> Result<Change> {
    if let Some(previous) = app_written_vault(agent)? {
        if previous != path_text(vault) {
            let old = PathBuf::from(previous);
            migrate_settings(writer, agent, &old, vault)?;
            migrate_instructions(writer, agent, &old, vault)?;
        }
    }
    let change = connect_server(writer, agent, exe, vault)?;
    if first_time {
        connect_settings(writer, agent, vault)?;
        connect_instructions(writer, agent, vault)?;
    }
    Ok(change)
}

/// Appends one line to `logs/<yyyy-MM>.jsonl` (WORKSPACE-SPEC 9): sorted keys, compact, slashes not escaped.
pub fn append(ws: &Workspace, kind: &str, actor: &str, subject: &str, outcome: &str, details: &[(&str, &str)], now: DateTime<Utc>) -> Result<()> {
    let mut sorted: Vec<&(&str, &str)> = details.iter().collect();
    sorted.sort_by_key(|(key, _)| *key);
    let mut detail_map = Map::new();
    for (key, value) in sorted {
        detail_map.insert((*key).to_string(), Value::String((*value).to_string()));
    }
    let mut entry = Map::new();
    entry.insert("actor".into(), actor.into());
    entry.insert("details".into(), Value::Object(detail_map));
    entry.insert("outcome".into(), outcome.into());
    entry.insert("subject".into(), subject.into());
    entry.insert("time".into(), iso(now).into());
    entry.insert("type".into(), kind.into());
    let mut line = serde_json::to_string(&Value::Object(entry))?;
    line.push('\n');

pub enum Step {

pub struct ConnectContext<'a> {
    pub ws: &'a Workspace,
    pub writer: &'a Writer,

    pub memory_exe: Option<&'a Path>,

pub struct StepFailure {
    pub agent_id: Uuid,
    pub agent_name: String,
    pub step: Step,
    pub message: String,
}

fn key(agent: &AgentProfile) -> String {
    agent.id.to_string().to_uppercase()
}

fn note(ctx: &ConnectContext, kind: &str, agent: &AgentProfile, extra: &[(&str, &str)]) {
    let id = key(agent);
    let mut details = vec![("agent", id.as_str())];
    details.extend_from_slice(extra);
    let _ = log::append(ctx.ws, kind, "app", &agent.name, "ok", &details, ctx.now);

/// Runs the four steps for every agent whose folder exists. First-time steps recorded in
/// `record` are skipped; the atrium-memory server entry is checked every time.
pub fn connect_all(ctx: &ConnectContext, agents: &[AgentProfile], record: &mut Connections) -> Vec<StepFailure> {
    let mut failures = Vec::new();
    for agent in agents.iter().filter(|a| a.path.is_dir()) {
        let failed = |step: Step, error: Error| StepFailure { agent_id: agent.id, agent_name: agent.name.clone(), step, message: error.to_string() };
        let done = record.agents.get(&key(agent)).cloned().unwrap_or_default();
        if done.inbox.is_none() {
            match connect_inbox(ctx.writer, ctx.ws, agent) {
                Ok(()) => {
                    record.agents.entry(key(agent)).or_default().inbox = Some(iso(ctx.now));
                    note(ctx, "agent.inbox-connected", agent, &[]);
                }
                Err(error) => failures.push(failed(Step::Inbox, error)),
            }
        }
        if done.widgets.is_none() {
            match connect_widgets(ctx.writer, ctx.ws, agent) {
                Ok(()) => {
                    record.agents.entry(key(agent)).or_default().widgets = Some(iso(ctx.now));
                    note(ctx, "agent.widgets-connected", agent, &[]);
                }
                Err(error) => failures.push(failed(Step::Widgets, error)),
            }
        }
        if let Some(exe) = ctx.memory_exe {
            let first_time = done.memory.is_none();
            match memory::connect(ctx.writer, agent, exe, &ctx.ws.vault(), first_time) {
                Ok(change) => {
                    if first_time {
                        record.agents.entry(key(agent)).or_default().memory = Some(iso(ctx.now));
                    }
                    if first_time || change != Change::Unchanged {
                        let change_text = format!("{change:?}").to_lowercase();
                        note(ctx, "agent.memory-connected", agent, &[("change", change_text.as_str())]);
                    }
                }
                Err(error) => failures.push(failed(if first_time { Step::MemorySetup } else { Step::MemoryServer }, error)),
            }
        }
    }
    failures
}

/// Runs one step in its first-time form, even when recorded (stage-1 design K1).
pub fn reconnect(ctx: &ConnectContext, agent: &AgentProfile, step: Step, record: &mut Connections) -> Result<()> {
    let vault = ctx.ws.vault();
    match step {
        Step::Inbox => connect_inbox(ctx.writer, ctx.ws, agent)?,
        Step::Widgets => connect_widgets(ctx.writer, ctx.ws, agent)?,
        Step::MemoryServer | Step::MemorySetup => {
            let Some(exe) = ctx.memory_exe else { return fail("The bundled atrium-memory server is missing.") };
            memory::connect_server(ctx.writer, agent, exe, &vault)?;
            if step == Step::MemorySetup {
                memory::connect_settings(ctx.writer, agent, &vault)?;
                memory::connect_instructions(ctx.writer, agent, &vault)?;
            }
        }
    }
    let entry = record.agents.entry(key(agent)).or_default();
    let stamp = Some(iso(ctx.now));
    match step {
        Step::Inbox => entry.inbox = stamp,
        Step::Widgets => entry.widgets = stamp,
        Step::MemoryServer | Step::MemorySetup => entry.memory = stamp,
    }
    Ok(())
}

use std::collections::HashMap;
use std::fs;
use std::path::{Path, PathBuf};

use serde_json::json;
use uuid::Uuid;

use crate::agent::{initial_files, AgentProfile, Presets, REQUIRED_DIRS};
use crate::fsutil::{write_atomic, Writer};
use crate::jsonedit;
use crate::paths::path_text;
use crate::{fail, Result};

#[derive(Debug, Clone)]
pub struct AgentEntry {
    pub folder: PathBuf,
    pub profile: Option<AgentProfile>,
    pub problems: Vec<String>,
    pub duplicate_id: bool,
}

const REQUIRED_FILES: [&str; 4] = ["CLAUDE.md", ".runtime/", "memory/MEMORY.md", ".mcp.json"];

/// Agents are folders in `agents_dir` (not starting with `.`) that contain agent.json, sorted by folder.
pub fn scan(agents_dir: &Path) -> Vec<AgentEntry> {
    let Ok(read) = fs::read_dir(agents_dir) else { return Vec::new() };
    let mut entries: Vec<AgentEntry> = read
        .flatten()
        .filter(|e| !e.file_name().to_string_lossy().starts_with('.') && e.path().join("agent.json").is_file())
        .map(|e| inspect(&e.path()))
        .collect();
    entries.sort_by(|a, b| a.folder.cmp(&b.folder));
    let mut counts: HashMap<Uuid, usize> = HashMap::new();
    for profile in entries.iter().filter_map(|e| e.profile.as_ref()) {
        *counts.entry(profile.id).or_default() += 1;
    }
    for entry in &mut entries {
        entry.duplicate_id = entry.profile.as_ref().is_some_and(|p| counts[&p.id] > 1);
    }
    entries
}

fn inspect(folder: &Path) -> AgentEntry {
    let mut problems = Vec::new();
    let parsed = fs::read_to_string(folder.join("agent.json"))
        .map_err(|e| e.to_string())
        .and_then(|text| AgentProfile::from_json(&text).map_err(|e| e.to_string()));
    let profile = match parsed {
        Ok(mut profile) => {
            profile.path = folder.to_path_buf();
            Some(profile)
        }
        Err(error) => {
            problems.push(format!("agent.json can't be read: {error}"));
            None
        }
    };
    for required in REQUIRED_FILES {
        let path = folder.join(required.trim_end_matches('/'));
        let present = if required.ends_with('/') { path.is_dir() } else { path.is_file() };
        if !present {
            problems.push(format!("{required} is missing"));
        }
    }
    AgentEntry { folder: folder.to_path_buf(), profile, problems, duplicate_id: false }
}

/// Creates missing folders and files with their initial contents; never overwrites (stage-1 design D2).
pub fn repair(entry: &AgentEntry) -> Result<()> {
    let Some(profile) = &entry.profile else {
        return fail("Fix agent.json first: Atrium needs the agent's name and ID to repair the rest.");
    };
    for child in REQUIRED_DIRS {
        fs::create_dir_all(entry.folder.join(child))?;
    }
    for (relative, text) in initial_files(profile, &Presets::default()) {
        let file = entry.folder.join(relative);
        if !file.exists() {
            write_atomic(&file, text.as_bytes())?;
        }
    }
    Ok(())
}

/// Gives a copied agent its own ID: new UUID in agent.json, folder suffix renamed, memory path updated (D3).
pub fn make_new_identity(writer: &Writer, entry: &AgentEntry) -> Result<AgentProfile> {
    let Some(old) = &entry.profile else { return fail("Fix agent.json first.") };
    let id = Uuid::new_v4();
    let old_suffix = format!("-{}", old.id.to_string().to_lowercase());
    let folder_name = entry.folder.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_default();
    let base = folder_name.strip_suffix("-copy").unwrap_or(&folder_name);
    let base = base.strip_suffix(&old_suffix).unwrap_or(base);
    let new_folder = entry.folder.with_file_name(format!("{base}-{}", id.to_string().to_lowercase()));
    fs::rename(&entry.folder, &new_folder)?;
    let profile = AgentProfile { id, path: new_folder.clone(), ..old.clone() };
    writer.write(&new_folder, &new_folder.join("agent.json"), &profile.to_json())?;
    let settings = new_folder.join(".runtime/settings.json");
    if let Ok(Some(mut doc)) = jsonedit::read(&settings) {
        let memory_key = doc.map.get("autoMemoryDirectory").and_then(|v| v.as_str()).map(str::to_string);
        let stale = memory_key.is_some_and(|value| value != path_text(&new_folder.join("memory")));
        if stale {
            doc.map.insert("autoMemoryDirectory".into(), json!(path_text(&new_folder.join("memory"))));
            writer.write(&new_folder, &settings, &doc.render())?;
        }
    }
    Ok(profile)
}

            findings.push(finding(Level::Warning, "skill-missing", format!("The {skill} skill is missing from the library"), "Reopen the workspace to restore it."));
        }
    }
    for entry in agents {
        let name = entry.profile.as_ref().map(|p| p.name.clone()).unwrap_or_else(|| path_text(&entry.folder));
        if !entry.problems.is_empty() {
            findings.push(finding(Level::Error, "agent-broken", format!("{name} needs repair"), entry.problems.join("; ")));
        }
        if entry.duplicate_id {
            findings.push(finding(Level::Error, "agent-duplicate", format!("{name} shares its ID with another agent"), path_text(&entry.folder)));
        }
    }
    if let Some(provider) = cloud_provider(&ws.root) {
        findings.push(finding(
            Level::Warning,
            "cloud-sync",
            format!("The workspace is inside {provider}"),
            "Sync can create conflict copies and online-only files. Keep the folder always available on this device.",
        ));
    }
    for folder in [ws.inbox(), ws.routines(), ws.records()] {
        let Ok(read) = fs::read_dir(&folder) else { continue };
        let mut names: Vec<String> = read.flatten().map(|e| e.file_name().to_string_lossy().into_owned()).filter(|n| is_conflict_copy(n)).collect();
        names.sort();
        for name in names {
            findings.push(finding(Level::Warning, "conflict-copy", format!("Sync conflict copy: {name}"), path_text(&folder.join(&name))));
        }
    }
    let mut skill_folders = vec![home.join(".claude/skills")];
    skill_folders.extend(agents.iter().map(|e| e.folder.join(".claude/skills")));
    for folder in skill_folders {
        for link in skills::dangling(&folder) {
            findings.push(finding(Level::Warning, "dangling-skill", "A skill link points nowhere", path_text(&link)));
        }
    }
    findings
}

#![cfg(unix)]

const FILES: [(&str, &str); 6] = [
    ("agent.json", "agent.json"),
    ("CLAUDE.md", "CLAUDE.md"),
    ("memory/MEMORY.md", "MEMORY.md"),
    (".mcp.json", "mcp.json"),
    (".runtime/settings.json", "settings.json"),
    (".claude/settings.local.json", "settings.local.json"),
];

fn fixtures() -> PathBuf {
    Path::new(env!("CARGO_MANIFEST_DIR")).join("tests/fixtures/claudedesk")
}

fn normalize(text: &str, ws: &Path, exe: &Path, id: &uuid::Uuid) -> String {
    let mut out = text.to_string();
    for (path, token) in [(exe, "{{EXE}}"), (ws, "{{WS}}")] {
        let plain = path.to_string_lossy().into_owned();
        out = out.replace(&plain.replace('/', "\\/"), token).replace(&plain, token);
    }
    out.replace(&id.to_string().to_uppercase(), "{{ID}}").replace(&id.to_string(), "{{id}}")
}

fn compare(stage: &str, agent_dir: &Path, ws: &Path, exe: &Path, id: &uuid::Uuid) {
    for (source, name) in FILES {
        let actual = normalize(&fs::read_to_string(agent_dir.join(source)).unwrap(), ws, exe, id);
        let expected = fs::read_to_string(fixtures().join(stage).join(name)).unwrap();
        assert_eq!(actual, expected, "{stage}/{name} differs from ClaudeDesk");
    }
}

#[test]
fn writes_what_claudedesk_writes() {
    let meta: serde_json::Value = serde_json::from_str(&fs::read_to_string(fixtures().join("meta.json")).unwrap()).unwrap();
